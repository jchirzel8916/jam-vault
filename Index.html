    function getColorHex(colorName) {

      if (!colorName) return '#9ca3af';

      const c = colorName.toLowerCase();

      if (c.includes('green') || c.includes('glow')) return '#10b981';

      if (c.includes('purple')) return '#a855f7';

      if (c.includes('black') || c.includes('bkt')) return '#1f2937';

      if (c.includes('chrome') || c.includes('silver')) return '#e5e7eb';

      if (c.includes('yellow') || c.includes('neon')) return '#facc15';

      if (c.includes('blue') || c.includes('ice')) return '#3b82f6';

      if (c.includes('red') || c.includes('flame')) return '#ef4444';

      if (c.includes('pink')) return '#ec4899';

      if (c.includes('teal')) return '#14b8a6';

      if (c.includes('gray')) return '#6b7280';

      return '#f59e0b';

    }

    function handleImageFile(e) {

      const file = e.target.files && e.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = function(event) {

        const img = new Image();

        img.onload = function() {

          // Downsample image for performant storage

          const maxDim = 800;

          let w = img.width;

          let h = img.height;

          if (w > maxDim || h > maxDim) {

            if (w > h) { h = (h / w) * maxDim; w = maxDim; }

            else { w = (w / h) * maxDim; h = maxDim; }

          }

          const canvas = document.createElement('canvas');

          canvas.width = w;

          canvas.height = h;

          const ctx = canvas.getContext('2d');

          ctx.drawImage(img, 0, 0, w, h);

          pendingImageData = canvas.toDataURL('image/jpeg', 0.85);

          drawImageOnCanvas(pendingImageData);

        };

        img.src = event.target.result;

      };

      reader.readAsDataURL(file);

    }

    function drawImageOnCanvas(src) {

      const wrapper = document.getElementById('canvas-wrapper');

      const canvas = document.getElementById('image-canvas');

      const ctx = canvas.getContext('2d');

      const img = new Image();

      img.onload = function() {

        wrapper.style.display = 'block';

        canvas.width = img.width;

        canvas.height = img.height;

        ctx.drawImage(img, 0, 0);

        // Click listener for color extraction

        canvas.onclick = function(evt) {

          if (!activePickerComponent) return;

          const rect = canvas.getBoundingClientRect();

          const scaleX = canvas.width / rect.width;

          const scaleY = canvas.height / rect.height;

          const x = (evt.clientX - rect.left) * scaleX;

          const y = (evt.clientY - rect.top) * scaleY;

          const pixel = ctx.getImageData(x, y, 1, 1).data;

          const colorName = approximateColorName(pixel[0], pixel[1], pixel[2]);

          if (activePickerComponent === 'chassis') {

            document.getElementById('input-chassis').value = colorName;

          } else if (activePickerComponent === 'rim') {

            document.getElementById('input-rim').value = colorName;

          } else if (activePickerComponent === 'tire') {

            document.getElementById('input-tire').value = colorName;

          }

          resetPickers();

          showToast(`Sampled ${colorName} for ${activePickerComponent}!`);

        };

      };

      img.src = src;

    }

    function approximateColorName(r, g, b) {

      // Basic heuristic for common monster truck finishes

      const max = Math.max(r, g, b);

      const min = Math.min(r, g, b);

      const diff = max - min;

      if (max < 50) return 'Black BKT';

      if (min > 200) return 'White';

      if (diff < 30) return 'Silver / Chrome';

      if (r > 150 && g < 80 && b < 80) return 'Red';

      if (g > 140 && r < 100) return 'Green Glow';

      if (b > 150 && r < 100) return 'Blue Ice';

      if (r > 140 && b > 140 && g < 100) return 'Purple';

      if (r > 180 && g > 150 && b < 80) return 'Neon Yellow';

      if (r > 180 && g > 90 && b < 60) return 'Orange';

      return `Custom (${r}, ${g}, ${b})`;

    }

        });

        client.requestAccessToken();

    function syncToGoogleDrive() {

      if (!gapiToken) {

        showToast('Local copy preserved (Sign in to sync with Google Drive)');

        return;

      }

      // Push file to Drive REST API v3

      fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=media', {

        method: 'POST',

        headers: {

          'Authorization': `Bearer ${gapiToken}`,

          'Content-Type': 'application/json'

        },

        body: JSON.stringify({

          app: 'JAM Vault',

          updatedAt: new Date().toISOString(),

          fleet: fleet

        })

      })

      .then(res => res.json())

      .then(() => showToast('Synced to Google Drive!'))

      .catch(() => showToast('Sync error. Local copy saved.'));

    }

      if (!gapiToken) {

        showToast('Please sign in to Google Drive first');

        return;

      }

      showToast('Searching Google Drive for jam_vault_backup.json...');

    }
